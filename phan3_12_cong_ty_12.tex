\subsection{Công ty 12: MB Bank (Ngân hàng TMCP Quân Đội)}

\textbf{Job Title:} Senior AI Engineer (Computer Vision/NLP/LLM) \\
\textbf{Link tuyển dụng:} \url{https://itviec.com/it-jobs/senior-ai-engineer-computer-vision-nlp-llm-mb-bank-2408}

\subsubsection*{Job Description (Nguyên gốc tiếng Việt)}

\textbf{Giới thiệu:} \\
\indent MB Bank là ngân hàng thương mại cổ phần Quân đội, một trong những ngân hàng lớn tại Việt Nam, đầu tư mạnh vào hạ tầng công nghệ và ứng dụng AI trong hoạt động ngân hàng.

\paragraph{Mô tả công việc:}
\begin{itemize}
    \item Nghiên cứu, phát triển, tối ưu và tích hợp các mô hình AI phục vụ nghiệp vụ trong lĩnh vực Computer Vision và NLP.
    \item Tham gia các giai đoạn triển khai dịch vụ: thiết kế kiến trúc hệ thống, thiết kế mô hình, tối ưu quy trình và hiệu năng.
    \item Thực hiện coding, kiểm thử theo metric và unit testing theo phương pháp luận DevOps và MLOps.
    \item Nghiên cứu các bài báo khoa học mới nhất về AI, Machine Learning, Deep Learning.
    \item Tham gia đào tạo, hướng dẫn sinh viên/thực tập sinh tài năng (đối với ứng viên fresher).
\end{itemize}

\paragraph{Yêu cầu công việc:}
\begin{itemize}
    \item Tốt nghiệp Đại học các ngành Công nghệ Thông tin, Khoa học Máy tính, Điện tử Viễn thông, Công nghệ Phần mềm hoặc liên quan.
    \item Tối thiểu 2 năm kinh nghiệm ở vị trí liên quan đến AI.
    \item Thành thạo ít nhất một trong các mảng: Python/C/C++; FastAPI/Flask/Django; TensorFlow/PyTorch/DGL/PyG/ONNX; RESTful/gRPC API, secure coding, tối ưu hiệu năng; Git, Docker, Kubernetes, Kubeflow/MLFlow, TritonServer.
    \item Có kiến thức về Supervised/Reinforcement Learning, Computer Vision, NLP hoặc xử lý âm thanh.
    \item Am hiểu các kiến trúc CNN, RNN, Transformer, Attention, SVM, GNN.
    \item Có kinh nghiệm về các kỹ thuật tối ưu mô hình: Knowledge Distillation, Pruning, Quantization.
\end{itemize}
